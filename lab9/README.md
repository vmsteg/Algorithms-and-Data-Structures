## Теоретическая часть

**Бинарное дерево** — иерархическая структура, в которой каждый узел имеет
не более двух потомков (левый и правый). Термины: корень, лист, внутренний
узел, высота дерева, уровень.

**Обходы бинарного дерева:**

| Вид обхода | Порядок посещения | Для дерева из примера |
| :--- | :--- | :--- |
| Прямой (pre-order) | Корень → Левый → Правый | 10, 5, 3, 7, 6, 15, 20, 18 |
| Симметричный (in-order) | Левый → Корень → Правый | 3, 5, 6, 7, 10, 15, 18, 20 |
| Обратный (post-order) | Левый → Правый → Корень | 3, 6, 7, 5, 18, 20, 15, 10 |
| Уровневый (level-order) | По уровням слева направо | 10, 5, 15, 3, 7, 20, 6, 18 |

**Бинарное дерево поиска (BST)** — бинарное дерево, для каждого узла которого
выполняется: все ключи левого поддерева < ключ узла < все ключи правого
поддерева. Симметричный обход BST даёт отсортированную последовательность.

| Операция | Средний | Худший | Память |
| :--- | :--- | :--- | :--- |
| insert | O(log n) | O(n) | O(1) |
| search | O(log n) | O(n) | O(1) |
| remove | O(log n) | O(n) | O(1) |
| обход | O(n) | O(n) | O(h) стек |

где **h** — высота дерева. Худший случай — вырожденное дерево (вставка
в отсортированном порядке), когда h = n.

## Листинг программы

```cpp
#include <iostream>
#include <queue>
#include <vector>
#include <cstdlib>
using namespace std;

// ===================== Узел =====================
struct Node {
    int data;
    Node* left;
    Node* right;
    Node(int v) : data(v), left(nullptr), right(nullptr) {}
};

// ===================== Обходы=====================
void preOrder(Node* n) {
    if (!n) return;
    cout << n->data << " ";
    preOrder(n->left);
    preOrder(n->right);
}

void inOrder(Node* n) {
    if (!n) return;
    inOrder(n->left);
    cout << n->data << " ";
    inOrder(n->right);
}

void postOrder(Node* n) {
    if (!n) return;
    postOrder(n->left);
    postOrder(n->right);
    cout << n->data << " ";
}

// Минимальный элемент
int findMin(Node* n) {
    if (!n) return 1e9;
    return min(n->data, min(findMin(n->left), findMin(n->right)));
}

// ===================== BST =====================
Node* insert(Node* n, int key) {
    if (!n) return new Node(key);
    if (key < n->data)      n->left  = insert(n->left, key);
    else if (key > n->data) n->right = insert(n->right, key);
    return n;
}

bool search(Node* n, int key) {
    if (!n) return false;
    if (n->data == key) return true;
    return key < n->data ? search(n->left, key) : search(n->right, key);
}

Node* findMinNode(Node* n) {
    while (n && n->left) n = n->left;
    return n;
}

Node* remove(Node* n, int key) {
    if (!n) return nullptr;
    if (key < n->data)      n->left  = remove(n->left, key);
    else if (key > n->data) n->right = remove(n->right, key);
    else {
        if (!n->left)  { Node* t = n->right; delete n; return t; }
        if (!n->right) { Node* t = n->left;  delete n; return t; }
        Node* m = findMinNode(n->right);
        n->data = m->data;
        n->right = remove(n->right, m->data);
    }
    return n;
}

int height(Node* n) {
    if (!n) return 0;
    return 1 + max(height(n->left), height(n->right));
}

// Преемник (наименьший ключ > key)
bool successor(Node* n, int key, int& res) {
    if (!n) return false;
    if (n->data > key) {
        res = n->data;
        successor(n->left, key, res);
        return true;
    }
    return successor(n->right, key, res);
}

void destroy(Node* n) {
    if (!n) return;
    destroy(n->left);
    destroy(n->right);
    delete n;
}

int main() {
    // ---- Демонстрация обходов ----
    Node* demo = new Node(10);
    demo->left = new Node(5);   demo->right = new Node(15);
    demo->left->left = new Node(3);   demo->left->right = new Node(7);
    demo->right->right = new Node(20);
    demo->left->right->left = new Node(6);
    demo->right->right->left = new Node(18);

    cout << "pre-order : "; preOrder(demo);  cout << "\n";
    cout << "in-order  : "; inOrder(demo);   cout << "\n";
    cout << "post-order: "; postOrder(demo); cout << "\n";
    cout << "min       : " << findMin(demo) << "\n\n";
    destroy(demo);

    // ---- BST из 20 случайных чисел ----
    srand(42);
    Node* root = nullptr;
    vector<int> used;
    while (used.size() < 20) {
        int v = rand() % 100 + 1;
        bool dup = false;
        for (int u : used) if (u == v) { dup = true; break; }
        if (!dup) { root = insert(root, v); used.push_back(v); }
    }

    cout << "in-order: "; inOrder(root); cout << "\n";
    // max — самый правый узел:
    Node* p = root; while (p->right) p = p->right;
    cout << "min = " << findMinNode(root)->data << ", max = " << p->data << "\n";
    cout << "height = " << height(root) << "\n";

    // ---- Преемник ----
    for (int k : {used[0], used[10], used[19]}) {
        int s;
        if (successor(root, k, s))
            cout << "successor(" << k << ") = " << s << "\n";
        else
            cout << "successor(" << k << ") = нет (максимум)\n";
    }

    // ---- Удаление ----
    int key;
    cout << "Введите значение для удаления: ";
    if (cin >> key) {
        if (search(root, key)) {
            root = remove(root, key);
            cout << "После удаления: "; inOrder(root); cout << "\n";
        } else {
            cout << "Элемент " << key << " не найден\n";
        }
    }

    destroy(root);
    return 0;
}
```

## Анализ сложности операций

| Операция | Средний | Худший | Комментарий |
| :--- | :--- | :--- | :--- |
| `insert` | O(log n) | O(n) | Вырожденное дерево при вставке по возрастанию |
| `search` | O(log n) | O(n) | То же |
| `remove` | O(log n) | O(n) | Поиск + нахождение минимума в правом поддереве |
| `preOrder`/`inOrder`/`postOrder` | O(n) | O(n) | Каждый узел посещается один раз |
| `findMin` (произвольное дерево) | O(n) | O(n) | Обход всех узлов |
| `findMinNode` (BST) | O(h) | O(n) | Спуск по левым потомкам |
| `successor` | O(h) | O(n) | Спуск по высоте h |
| `height` | O(n) | O(n) | Обход всех узлов |

где **h** — высота дерева: h ≈ log n для сбалансированного, h = n для
вырожденного.

## Ответы на контрольные вопросы

1. **Бинарное дерево** — иерархическая структура, каждый узел имеет не более
   двух потомков. Состоит из корня, внутренних узлов и листьев; каждый узел
   хранит данные и указатели на левого/правого потомка.

2. **Отличие BST от бинарного дерева:** в BST для каждого узла выполняется
   `все ключи левого поддерева < ключ узла < все ключи правого поддерева`.
   В обычном бинарном дереве такого условия нет.

3. **Виды обхода:**
   - pre-order (корень → лево → право) — копирование дерева;
   - in-order (лево → корень → право) — сортированная последовательность для BST;
   - post-order (лево → право → корень) — удаление дерева, вычисление выражений;
   - level-order (BFS) — поиск в ширину.

4. **Сортированную последовательность для BST даёт in-order обход.**

5. **Сложность операций BST:** O(log n) в среднем (сбалансированное дерево),
   O(n) в худшем (вырожденное). Зависит от высоты дерева.

6. **При вставке по возрастанию** BST вырождается в связный список,
   высота = n, все операции O(n). Защита — балансировка (AVL, красно-чёрные
   деревья).

7. **Три случая удаления:**
   - лист — просто удаляем;
   - один потомок — заменяем узел его потомком;
   - два потомка — находим минимум в правом поддереве (или максимум
     в левом), копируем значение и удаляем тот узел.

8. **Минимум** — самый левый узел; **максимум** — самый правый.

9. **Высота рекурсивно:** `height(n) = 1 + max(height(left), height(right))`,
   база — `height(nullptr) = 0`.

10. **LCA** — самый глубокий узел, являющийся предком обоих заданных узлов.
    В BST находится за O(h): идём от корня, пока оба ключа не окажутся
    по разные стороны.

11. **Проверка BST:** рекурсивно проверяем, что `left->data < node->data <
    right->data` и что оба поддерева сами являются BST с учётом диапазона
    `(min, max)`. Альтернатива — in-order обход должен быть строго
    возрастающим.

12. **Сбалансированное дерево** — высоты левого и правого поддеревьев
    отличаются не более чем на 1 (AVL) или выполняется красно-чёрное свойство.
    Важно, потому что от высоты зависит сложность операций: O(log n) при
    балансе, O(n) без него.

13. **Преобразование в сбалансированное:** получить in-order обход
    (отсортированный массив), затем рекурсивно строить дерево из середины
    массива — O(n).

14. **Восстановление по in-order + pre-order:** первый элемент pre-order —
    корень; в in-order находим его позицию, левая часть — левое поддерево,
    правая — правое. Рекурсивно строим.

15. **Итеративный обход** — с явным стеком (для pre/in/post-order) или
    очередью (для level-order). Pre-order: кладём корень в стек, в цикле
    извлекаем, печатаем, кладём сначала правого, потом левого потомка.
